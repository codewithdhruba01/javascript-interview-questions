2. Prototypes & Inheritance

// Constructor function function Animal(name) { this.name = name; } // Add method to prototype Animal.prototype.speak = function() { return this.name + ' makes a noise.'; }; // Inheritance function Dog(name, breed) { Animal.call(this, name); this.breed = breed; } Dog.prototype = Object.create(Animal.prototype); Dog.prototype.constructor = Dog; Dog.prototype.speak = function() { return this.name + ' barks.'; };

3. Type Coercion

// Implicit coercion console.log(1 + '2'); // '12' (number to string) console.log('1' + 2); // '12' (number to string) console.log(1 - '2'); // -1 (string to number) console.log('5' * 2); // 10 (string to number) console.log('5' == 5); // true (loose equality) console.log('5' === 5); // false (strict equality) // Explicit conversion console.log(Number('123')); // 123 console.log(String(123)); // '123' console.log(Boolean(0)); // false console.log(Boolean('')); // false

5. Memory Management

⚠️ Memory Leak Demo:
Closures can hold references to large objects
// Potential memory leak function createLeak() { const largeData = new Array(100000).fill('data'); return function() { console.log('Still holding', largeData.length, 'items'); }; } // Proper cleanup function createNoLeak() { let largeData = new Array(100000).fill('data'); return function() { console.log('Processing data...'); largeData = null; // Allow GC }; }

6. Error Handling

// Try-catch-finally function riskyOperation() { if (Math.random() > 0.5) { throw new Error('Something went wrong!'); } return 'Success!'; } try { const result = riskyOperation(); console.log('Result:', result); } catch (error) { console.error('Caught error:', error.message); } finally { console.log('Cleanup always runs'); } // Custom error class class ValidationError extends Error { constructor(message, field) { super(message); this.name = 'ValidationError'; this.field = field; } }
